### 需要新增的系统程序扫描
```rust
#[tauri::command]
pub fn scan_system_programs() -> Result<Vec<LaunchItem>, String> {
    let mut programs = Vec::new();
    
    // Windows: 扫描常见路径
    #[cfg(target_os = "windows")]
    {
        scan_windows_programs(&mut programs)?;
    }
    
    // 为每个程序提取图标
    for program in &mut programs {
        if let Some(icon) = extract_icon_base64(&program.target_path) {
            program.icon_base64 = Some(icon);
        }
    }
    
    Ok(programs)
}

fn scan_windows_programs(programs: &mut Vec<LaunchItem>) -> Result<(), String> {
    use std::fs;
    use glob::glob;
    
    // 扫描路径列表
    let scan_paths = vec![
        "C:/Program Files/**/*.exe",
        "C:/Program Files (x86)/**/*.exe",
        "C:/Windows/System32/*.exe",
        // 用户桌面
        &format!("{}Desktop/*.lnk", dirs::home_dir().unwrap().to_str().unwrap()),
    ];
    
    for pattern in scan_paths {
        for entry in glob(pattern).map_err(|e| e.to_string())? {
            if let Ok(path) = entry {
                let name = path.file_stem()
                    .and_then(|s| s.to_str())
                    .unwrap_or("Unknown")
                    .to_string();
                    
                programs.push(LaunchItem {
                    name,
                    target_path: path.to_string_lossy().to_string(),
                    icon_location: String::new(),
                    arguments: String::new(),
                    count: 0,
                    icon_base64: None,
                });
            }
        }
    }
    
    Ok(())
}
```

### 前端数据处理（利用现有配置系统）
```typescript
// 程序启动（支持管理员模式）
const launchProgram = async (item: LaunchItem, runAsAdmin = false) => {
  try {
    await invoke('launch_program', {
      req: {
        target_path: item.target_path,
        arguments: item.arguments || null,
        run_as_admin: runAsAdmin
      }
    })
    
    // 更新使用统计
    await updateUsageCount(item.name)
    ElMessage.success(`启动 ${item.name}`)
  } catch (error) {
    ElMessage.error('启动程序失败')
  }
}

// 获取配置并显示图标
const loadLaunchConfig = async () => {
  try {
    const config = await invoke('get_all_launch_items') as LaunchConfig
    
    // 为没有图标的项目加载图标
    for (const category in config) {
      for (const item of config[category]) {
        if (!item.icon_base64) {
          try {
            const iconData = await invoke('get_icon_base64', { 
              path: item.target_path 
            })
            item.icon_base64 = iconData
          } catch (e) {
            // 忽略图标加载失败
          }
        }
      }
    }
    
    launcherStore.setConfig(config)
  } catch (error) {
    ElMessage.error('加载配置失败')
  }
}
```
