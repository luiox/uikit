```rust
#[tauri::command]
pub fn add_launch_item(category: String, item: LaunchItem) -> Result<(), String> {
    let mut config = load_config(CONFIG_PATH).map_err(|e| e.to_string())?;
    
    // 获取图标
    let mut new_item = item;
    if let Some(icon) = extract_icon_base64(&new_item.target_path) {
        new_item.icon_base64 = Some(icon);
    }
    
    config.entry(category).or_insert_with(Vec::new).push(new_item);
    save_config_file(&config, CONFIG_PATH).map_err(|e| e.to_string())?;
    Ok(())
}

#[tauri::command]
pub fn remove_launch_item(name: String, category: String) -> Result<(), String> {
    let mut config = load_config(CONFIG_PATH).map_err(|e| e.to_string())?;
    
    if let Some(items) = config.get_mut(&category) {
        items.retain(|item| item.name != name);
        if items.is_empty() {
            config.remove(&category);
        }
    }
    
    save_config_file(&config, CONFIG_PATH).map_err(|e| e.to_string())?;
    Ok(())
}
```

```vue
<script setup lang="ts">
import { invoke } from '@tauri-apps/api/tauri'
import type { LaunchConfig, LaunchItem, LaunchProgramRequest } from '@/types/launcher'

const searchKeyword = ref('')
const selectedCategory = ref('all')
const launchConfig = ref<LaunchConfig>({})
const currentItem = ref<LaunchItem | null>(null)

// 计算属性
const categories = computed(() => Object.keys(launchConfig.value))

const allPrograms = computed(() => {
  const programs: Array<LaunchItem & { category: string }> = []
  for (const [category, items] of Object.entries(launchConfig.value)) {
    items.forEach(item => {
      programs.push({ ...item, category })
    })
  }
  return programs.sort((a, b) => b.count - a.count) // 按使用次数排序
})

const filteredPrograms = computed(() => {
  let programs = allPrograms.value
  
  // 分类筛选
  if (selectedCategory.value !== 'all') {
    programs = programs.filter(p => p.category === selectedCategory.value)
  }
  
  // 搜索筛选
  if (searchKeyword.value) {
    programs = programs.filter(p => 
      p.name.toLowerCase().includes(searchKeyword.value.toLowerCase()) ||
      p.target_path.toLowerCase().includes(searchKeyword.value.toLowerCase())
    )
  }
  
  return programs
})

// 方法
const loadConfig = async () => {
  try {
    const config = await invoke('get_all_launch_items') as LaunchConfig
    launchConfig.value = config
  } catch (error) {
    ElMessage.error('加载配置失败')
  }
}

const launchProgram = async (item: LaunchItem, runAsAdmin: boolean = false) => {
  try {
    const request: LaunchProgramRequest = {
      target_path: item.target_path,
      arguments: item.arguments || undefined,
      run_as_admin: runAsAdmin

    await invoke('launch_program', { req: request })
    
    // 更新使用次数（可选实现）
    await updateUsageCount(item, runAsAdmin ? '管理员' : '普通')
    
    ElMessage.success(`${runAsAdmin ? '以管理员权限' : ''}启动 ${item.name}`)
  } catch (error) {
    ElMessage.error(`启动失败: ${error}`)
  }
}

const updateUsageCount = async (item: LaunchItem, mode: string) => {
  // 这里可以实现使用统计更新逻辑
  console.log(`${item.name} 以${mode}模式启动`)
}

const showContextMenu = (event: MouseEvent, item: LaunchItem) => {
  currentItem.value = item
  // 实现右键菜单显示逻辑
}

const handleContextMenuCommand = async (command: string) => {
  if (!currentItem.value) return
  
  switch (command) {
    case 'launch':
      await launchProgram(currentItem.value, false)
      break
    case 'launch-admin':
      await launchProgram(currentItem.value, true)
      break
    case 'edit':
      // 实现编辑逻辑
      break
    case 'delete':
      await deleteItem(currentItem.value)
      break
  }
}

const deleteItem = async (item: LaunchItem) => {
  try {
    await ElMessageBox.confirm(`确定删除 ${item.name}？`, '确认删除')
    
    await invoke('remove_launch_item', {
      name: item.name,
      category: item.category
    })
    
    await loadConfig() // 重新加载配置
    ElMessage.success('删除成功')
  } catch (error) {
    if (error !== 'cancel') {
      ElMessage.error('删除失败')
    }
  }
}

// 生命周期
onMounted(() => {
  loadConfig()
})
</script>
```

```rust
pub fn launch_program(path: String) -> Result<(), String> {
```
